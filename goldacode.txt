import re

from infra.cmx.cmx import Cmx as CMX
from infra.tfls.base_tfl import BaseTfl

class BitbucketImportOps(BaseTfl):

    def __init__(self, cmx: CMX = None, tfls=None, logger=None):
        super().__init__(cmx=cmx, logger=logger, tfls=tfls)

    @BaseTfl.step_wrapper(title="step_import_project_via_bitbucket_wizard",
                          error="Failed to import project via Bitbucket wizard")
    def step_import_project_via_bitbucket_wizard(self, instance_name: str, bitbucket_url: str, token: str,
                                                 org_name: str, repo_name: str, **kwargs):
        """
        Import a project through the Bitbucket UI wizard: New Project > Code Repository Integration >
        Bitbucket > add configuration > select organization > select repository > advance through wizard.
        :return: None
        """
        self.logger.info(f"Importing project via Bitbucket wizard: instance {instance_name}, org {org_name}, repo {repo_name}")
        page = self.cmx.ui.page
        page.goto(f"{self.cmx.environment.server_url}/applicationsAndProjects/projects", wait_until="domcontentloaded")
        page.get_by_role("button", name="New", exact=True).click()
        page.get_by_role("menuitem", name=re.compile("New Project - Code Repository")).click()
        self._handle_custom_setup_if_present(page)
        page.get_by_role("radio", name=re.compile(r"^Bitbucket, \d+ of")).click()
        page.get_by_role("button", name="+ Add Configuration").click()
        page.get_by_role("textbox", name="* Instance Name").fill(instance_name)
        page.get_by_role("textbox", name="* URL").fill(bitbucket_url)
        page.get_by_role("textbox", name="* Token").fill(token)
        page.get_by_role("button", name="Save & Continue").click()
        page.get_by_role("textbox", name="Search").fill("rn")
        self._check_or_click_org(page, org_name)
        page.get_by_role("button", name="Select Organization").click()
        self._check_or_click_repo(page, repo_name)
        page.get_by_role("button", name="Select Repositories").click()
        self._advance_through_wizard_until_create(page)
        page.expect_to_have_text(page.get_by_role("alert").first, re.compile(".+"), timeout=60_000)
        self.logger.info("   import started: " + page.get_by_role("alert").first.inner_text().replace("\n", " ")[:120])

    def _handle_custom_setup_if_present(self, page):
        """
        Click Custom Setup if it appears (some environments have it, others do not).
        :return: None
        """
        try:
            custom_setup = page.get_by_text("Custom Setup")
            custom_setup.wait_for(state="visible", timeout=5_000)
            custom_setup.click()
            self.logger.info("   selected Custom Setup option")
        except Exception as err:
            self.logger.info(f"   no Custom Setup option on this environment: {str(err).splitlines()[0][:100]}")

    def _check_or_click_org(self, page, org_name):
        """
        Select the organization by checking it (checkbox) or clicking it (label), whichever is present.
        :return: None
        """
        try:
            page.get_by_text(org_name, exact=True).check(timeout=5_000)
        except Exception as err:
            self.logger.info(f"   org checkbox not available, clicking text: {str(err).splitlines()[0][:100]}")
            page.get_by_text(org_name, exact=True).click()

    def _check_or_click_repo(self, page, repo_name):
        """
        Select the repository by checking it (checkbox) or clicking it (label), whichever is present.
        :return: None
        """
        try:
            page.get_by_text(repo_name).check(timeout=5_000)
        except Exception as err:
            self.logger.info(f"   repo checkbox not available, clicking text: {str(err).splitlines()[0][:100]}")
            page.get_by_text(repo_name).click()

    def _advance_through_wizard_until_create(self, page):
        """
        Keep advancing through wizard steps (Next / Create Project) until the final alert appears.
        :return: None
        """
        next_btn = page.get_by_role("button", name="Next", exact=True)
        create_btn = page.get_by_role("button", name="Create Project")
        alert = page.get_by_role("alert")
        repo_checks = page.locator("#code-repository-flows-mfe-1769006270777-3")
        for _ in range(10):
            if create_btn.is_visible():
                create_btn.click()
                break
            if next_btn.is_visible():
                if not next_btn.is_enabled():
                    self._try_enable_next_via_repo_checks(repo_checks, next_btn)
                next_btn.click()
            if alert.first.is_visible():
                break
            page.wait_for_timeout(1_500)

    def _try_enable_next_via_repo_checks(self, repo_checks, next_btn):
        """
        Attempt to enable the Next button by clicking repository checkboxes.
        :return: None
        """
        for idx in (1, 0, -1):
            try:
                repo_checks.nth(idx).click(timeout=3_000)
            except Exception:
                continue
            if next_btn.is_enabled():
                break

    def choose_dropdown(page, trigger, option_text, attempts=3):
        """
        Custom dropdown helper: click the control, wait for the option list, click the option.
        :return: None
        """
        try:
            option_locator = (page.get_by_role("option", name=option_text)
                              .or_(page.get_by_title(option_text, exact=True))
                              .filter(visible=True).first)
            for _ in range(attempts):
                trigger.click()
                try:
                    option_locator.wait_for(state="visible", timeout=4_000)
                    option_locator.click()
                    return
                except Exception:
                    page.keyboard.press("Escape")
            raise RuntimeError(f"Dropdown option {option_text!r} did not open")
        except Exception as err:
            raise RuntimeError(f"choose_dropdown failed: {str(err).splitlines()[0][:150]}")

    @BaseTfl.step_wrapper(title="step_wait_for_new_scan_with_origin", error="Failed to wait for new scan with origin")
    def step_wait_for_new_scan_with_origin(self, project_id: str, branch: str, before_count: int, expected_origin: str,
                                           timeout: int, **kwargs):
        """
        Poll until the scan list for the project/branch has more scans than before_count and the newest scan's
        origin matches the expected_origin, then wait for that scan to reach Completed status.
        :return: None
        """
        self.logger.info(f"Waiting for a new scan on project {project_id}, branch {branch} with origin {expected_origin}")
        deadline = time.time() + timeout
        while time.time() < deadline:
            scans_response = self.cmx.rest.scans.get_list_of_scans(params={"project-id": project_id, "branch": branch,
                                                                            "limit": 10, "offset": 0})
            if scans_response.status_code not in [200, 201]:
                time.sleep(15)
                continue
            scans_data = scans_response.json()
            scans_list = scans_data.get("scans", [])
            total_count = scans_data.get("totalCount", 0)
            if total_count > before_count and scans_list:
                newest_scan = scans_list[0]
                scan_origin = newest_scan.get("origin", "").lower()
                if expected_origin.lower() in scan_origin:
                    scan_id = newest_scan.get("id")
                    self.logger.info(f"   found new scan {scan_id} with origin {expected_origin}")
                    self.tfls.BE.performance.step_wait_for_scan_to_be_in_status(project_id=project_id, branch=branch,
                                                                                 expected_status=["Completed"],
                                                                                 timeout=timeout)
                    return
            time.sleep(15)
        raise TimeoutError(f"No new completed scan with origin {expected_origin} within {timeout}s")

    @BaseTfl.step_wrapper(title="step_push_file_to_bitbucket", error="Failed to push file to Bitbucket")
    def step_push_file_to_bitbucket(self, bb_host: str, project_key: str, repo_slug: str, filename: str, content: str,
                                    branch: str, commit_message: str, **kwargs):
        """
        Push a file to the Bitbucket repository on the given branch via the REST API.
        :return: None
        """
        self.logger.info(f"Pushing file {filename} to {branch} in {project_key}/{repo_slug}")
        api_base = f"http://{bb_host}/rest/api/1.0/projects/{project_key}/repos/{repo_slug}"
        headers = bb_headers(self.cmx.environment.bitbucket_user, self.cmx.environment.bitbucket_password)
        api = self.cmx.ui.page.request
        res = api.put(f"{api_base}/browse/{filename}", headers=headers,
                      multipart={"content": {"name": filename, "mimeType": "text/plain", "buffer": content.encode()},
                                 "message": commit_message, "branch": branch})
        if res.status not in (200, 201):
            raise RuntimeError(f"Push to {branch} failed: HTTP {res.status} {res.text()[:300]}")

    @BaseTfl.step_wrapper(title="step_create_pull_request", error="Failed to create pull request")
    def step_create_pull_request(self, bb_host: str, project_key: str, repo_slug: str, from_branch: str, to_branch: str,
                                 title: str, **kwargs):
        """
        Create a pull request from from_branch to to_branch in the Bitbucket repository.
        :return: pull request ID
        """
        self.logger.info(f"Creating PR from {from_branch} to {to_branch} in {project_key}/{repo_slug}")
        api_base = f"http://{bb_host}/rest/api/1.0/projects/{project_key}/repos/{repo_slug}"
        headers = bb_headers(self.cmx.environment.bitbucket_user, self.cmx.environment.bitbucket_password)
        body = {"title": title, "description": "automated", "fromRef": ref(from_branch, repo_slug, project_key),
                "toRef": ref(to_branch, repo_slug, project_key)}
        api = self.cmx.ui.page.request
        res = api.post(f"{api_base}/pull-requests", headers={**headers, "Content-Type": "application/json"},
                       data=json.dumps(body))
        if res.status not in [200, 201]:
            raise RuntimeError(f"Create PR failed: HTTP {res.status} {res.text()[:300]}")
        pr_id = res.json()["id"]
        self.logger.info(f"   created PR #{pr_id}")
        return pr_id

    @BaseTfl.step_wrapper(title="step_merge_pull_request", error="Failed to merge pull request")
    def step_merge_pull_request(self, bb_host: str, project_key: str, repo_slug: str, pr_id: int, **kwargs):
        """
        Merge the given pull request in the Bitbucket repository.
        :return: None
        """
        self.logger.info(f"Merging PR #{pr_id} in {project_key}/{repo_slug}")
        api_base = f"http://{bb_host}/rest/api/1.0/projects/{project_key}/repos/{repo_slug}"
        headers = bb_headers(self.cmx.environment.bitbucket_user, self.cmx.environment.bitbucket_password)
        api = self.cmx.ui.page.request
        detail = api.get(f"{api_base}/pull-requests/{pr_id}", headers=headers)
        version = detail.json()["version"]
        res = api.post(f"{api_base}/pull-requests/{pr_id}/merge?version={version}", headers=headers)
        if res.status not in [200, 201]:
            raise RuntimeError(f"Merge PR failed: HTTP {res.status} {res.text()[:300]}")
        self.logger.info(f"   merged PR #{pr_id}")

    def ref(branch, repo_slug, project_key):
        """
        Build a Bitbucket ref object for the given branch.
        :return: dict with ref structure
        """
        return {"id": f"refs/heads/{branch}", "repository": {"slug": repo_slug, "project": {"key": project_key}}}

    def bb_headers(user, password):
        """
        Build the Authorization header for Bitbucket REST API calls.
        :return: dict with Authorization header
        """
        try:
            raw = f"{user}:{password}".encode()
            return {"Authorization": "Basic " + base64.b64encode(raw).decode(), "X-Atlassian-Token": "no-check"}
        except Exception as err:
            raise RuntimeError(f"bb_headers failed: {str(err).splitlines()[0][:150]}")

    def list_open_prs(api, api_base, headers, from_branch, to_branch):
        """
        List open PRs from from_branch to to_branch.
        :return: list of matching PRs or None if failed
        """
        try:
            res = api.get(f"{api_base}/pull-requests?state=OPEN&limit=50", headers=headers)
            if res.status not in [200, 201]:
                return None
            return [pr for pr in res.json().get("values", [])
                    if pr["fromRef"]["displayId"] == from_branch and pr["toRef"]["displayId"] == to_branch]
        except Exception as err:
            raise RuntimeError(f"list_open_prs failed: {str(err).splitlines()[0][:150]}")

    def decline_pr(api, api_base, headers, pr, logger):
        """
        Decline a single PR.
        :return: None
        """
        try:
            api.post(f"{api_base}/pull-requests/{pr['id']}/decline?version={pr['version']}", headers=headers)
            logger.info(f"   declined leftover PR #{pr['id']}")
        except Exception as err:
            logger.info(f"   (could not decline PR #{pr['id']}: {str(err).splitlines()[0][:140]})")
